Step 1: Security Guard Banana (Middleware)


1. backend folder ke andar ek naya folder banwayein middleware aur usme ek file banayein authMiddleware.js.

2.  Ye guard har request aane par Token verify karega.

----------------------------------------------------------------------------------------------------------
backend/middleware/authMiddleware.js
-----------------------------------------

const jwt = require('jsonwebtoken');

const protect = (req, res, next) => {
  let token = req.headers.authorization; 

  if (!token || !token.startsWith('Bearer')) {
    return res.status(401).json({ message: "Bina Login ke aap andar nahi aa sakte! (No Token)" });
  }

  try {
    // "Bearer eyJhbGc..." mein se sirf token nikalna
    token = token.split(' ')[1];
    
    // Secret key se token ko verify karna
    const decoded = jwt.verify(token, process.env.JWT_SECRET);
    
    // Request ke andar user ki asali ID attach kar dena
    req.user = decoded; 
    
    next(); // Security check pass! Aage Controller ke paas jao
  } catch (error) {
    res.status(401).json({ message: "Token Invalid ya Expire ho chuka hai!" });
  }
};

module.exports = { protect };

const mongoose = require('mongoose');

const watchlistSchema = new mongoose.Schema({
  user: {
    type: mongoose.Schema.Types.ObjectId,
    ref: 'User', // Ye 'User' table se link hai
    required: true
  },
  movie: {
    type: mongoose.Schema.Types.ObjectId,
    ref: 'Movie', // Ye 'Movie' table se link hai
    required: true
  }
}, { timestamps: true });

module.exports = mongoose.model('Watchlist', watchlistSchema);

Step 3: Watchlist Logic (controllers/watchlistController.js)

const Watchlist = require('../models/Watchlist');
-----------------------------------------------------
// 1. Apni Watchlist ki saari movies mangwana (GET)
------------------------------------------------------
const getWatchlist = async (req, res) => {
  try {
    // .populate('movie') ka matlab hai Movie ki sirf ID nahi, poori details (Poster, Title) bhej do!
    const watchlist = await Watchlist.find({ user: req.user.id }).populate('movie');
    res.status(200).json(watchlist);
  } catch (error) {
    res.status(500).json({ message: "Server Error", error });
  }
};
----------------------------------------------------------------
// 2. Watchlist mein nayi movie daalna (POST)
----------------------------------------------------------------
const addToWatchlist = async (req, res) => {
  try {
    const { movieId } = req.body;

    // Check karein ki user ne ye movie pehle se toh save nahi kar rakhi
    const alreadySaved = await Watchlist.findOne({ user: req.user.id, movie: movieId });
    if (alreadySaved) return res.status(400).json({ message: "Movie already in watchlist!" });

    const watchlistItem = new Watchlist({ user: req.user.id, movie: movieId });
    await watchlistItem.save();
    
    res.status(201).json({ message: "Movie added to watchlist!", watchlistItem });
  } catch (error) {
    res.status(500).json({ message: "Server Error", error });
  }
};

module.exports = { getWatchlist, addToWatchlist };
